fn generate_synthetic_point_target(
    n_az: usize,
    n_rg: usize,
    target_az: usize,
    target_rg: usize,
) -> Array2<Complex32> {
    use num_complex::Complex32;
    use std::f32::consts::PI;

    let mut data = Array2::<Complex32>::zeros((n_az, n_rg));
    let bandwidth = 20.0e6_f32;
    let sample_rate = 25.0e6_f32;
    let prf = 1600.0_f32;
    let chirp_rate = bandwidth / 50.0e-6_f32;

    for az in 0..n_az {
        for rg in 0..n_rg {
            let t = (rg as f32 - target_rg as f32) / sample_rate;
            let eta = (az as f32 - target_az as f32) / prf;

            let range_chirp = Complex32::from_polar(1.0, PI * chirp_rate * t * t);
            let az_phase = Complex32::from_polar(1.0, -PI * 1000.0 * eta * eta);
            let envelope_r = (-((t * sample_rate).powi(2)) / (2.0 * 20.0_f32.powi(2))).exp();
            let envelope_az = (-((eta * prf).powi(2)) / (2.0 * 50.0_f32.powi(2))).exp();

            data[[az, rg]] = range_chirp * az_phase * envelope_r * envelope_az;
        }
    }
    info!(
        "Synthetic point target at [{}, {}] in {} × {} scene",
        target_az, target_rg, n_az, n_rg
    );
    data
}

/// Apply Range Cell Migration Correction to range-Doppler domain data
///
/// Now uses full sinc interpolation (your beautiful kernel!) instead of linear lerp.
pub fn apply_rcmc(
    range_doppler_data: &Array2<Complex32>,
    sample_rate: f32,
    prf: f32,
    wavelength: f32,
    velocity: f32,
    near_range: f32,
) -> Array2<Complex32> {
    let (num_range, num_doppler) = range_doppler_data.dim();
    info!(
        "Applying RCMC with SINC interpolation: {}×{}, λ={:.3}m, v={:.0}m/s",
        num_range, num_doppler, wavelength, velocity
    );

    let mut corrected = Array2::zeros((num_range, num_doppler));

    // Range sample spacing in meters
    let c = 299792458.0_f32;
    let range_spacing = c / (2.0 * sample_rate);

    // Process column-by-column (fixed Doppler frequency) — this is the SAR-standard way
    for doppler_idx in 0..num_doppler {
        // Doppler frequency (centered)
        let doppler_freq = ((doppler_idx as f32) - (num_doppler as f32 / 2.0)) * prf / (num_doppler as f32);

        // Extract the full range line for this Doppler bin
        let range_line: Vec<Complex32> = (0..num_range)
            .map(|r| range_doppler_data[[r, doppler_idx]])
            .collect();

        // Use center range for shift calculation (excellent approximation)
        let center_range_idx = num_range / 2;
        let slant_range = near_range + (center_range_idx as f32) * range_spacing;
        let migration_meters = range_migration(doppler_freq, wavelength, velocity, slant_range);
        let migration_samples = migration_meters / range_spacing;

        // Shift the entire range line with your sinc kernel (opposite direction to correct)
        let shifted_line = sinc_interpolate_shift(&range_line, -migration_samples);

        // Write back to corrected matrix
        for r in 0..num_range {
            corrected[[r, doppler_idx]] = shifted_line[r];
        }
    }

    info!("RCMC with sinc interpolation complete.");
    corrected
}

    base_url: String,
    token: Option<String>,
    token_expiry: Option<Instant>,
}

impl EsaClient {
    pub fn new() -> Self {
        Self {
            client: Client::new(),
            base_url: "https://catalogue.dataspace.copernicus.eu/odata/v1".to_string(),
            token: None,
            token_expiry: None,
        }
    }

    pub async fn get_access_token(&mut self) -> Result<String, String> {
        if let Some(token) = &self.token {
            if let Some(expiry) = self.token_expiry {
                if expiry > Instant::now() {
                    return Ok(token.clone());
                }
            }
        }

        let username = env::var("ESA_USERNAME").map_err(|_| "ESA_USERNAME not set")?;
        let password = env::var("ESA_PASSWORD").map_err(|_| "ESA_PASSWORD not set")?;

        let params = [
            ("client_id", "cdse-public"),
            ("username", &username),
            ("password", &password),
            ("grant_type", "password"),
        ];

        let resp = self.client
            .post("https://identity.dataspace.copernicus.eu/auth/realms/CDSE/protocol/openid-connect/token")
            .form(&params)
            .send()
            .await
            .map_err(|e| e.to_string())?;

        if !resp.status().is_success() {
            let err = resp.text().await.unwrap_or_default();
            error!("ESA Auth Error: {}", err);
            return Err("Failed to authenticate with ESA".into());
        }

        let json: serde_json::Value = resp.json().await.map_err(|e| e.to_string())?;
        let token = json["access_token"].as_str().ok_or("No access_token")?.to_string();

        self.token = Some(token.clone());
        self.token_expiry = Some(Instant::now() + Duration::from_secs(3600)); // 1 hour

        Ok(token)
    }

    pub async fn search_scenes(
        &mut self,
        lat: f64,
        lon: f64,
        start_date: Option<String>,
        end_date: Option<String>,
    ) -> Result<Vec<SarScene>, String> {
        let token = self.get_access_token().await?;

        let mut filter = format!(
            "OData.CSC.Intersects(area=geography'SRID=4326;POINT({} {})') and Collection/Name eq 'SENTINEL-1'",
            lon, lat
        );

        filter.push_str(" and Attributes/OData.CSC.StringAttribute/any(i:i/Name eq 'productType' and i/Value eq 'GRD')");

        if let Some(start) = start_date {
            filter.push_str(&format!(" and ContentDate/Start ge {}", start));
        }
        if let Some(end) = end_date {
            filter.push_str(&format!(" and ContentDate/End le {}", end));
        }

        let url = format!(
            "{}/Products?$filter={}&$top=10&$orderby=ContentDate/Start desc&$expand=Attributes",
            self.base_url, filter
        );

        info!("Querying ESA: {}", url);

        let resp = self.client.get(&url).bearer_auth(token).send().await.map_err(|e| e.to_string())?;

        if !resp.status().is_success() {
            let err = resp.text().await.unwrap_or_default();
            error!("ESA API Error: {}", err);
            return Err("ESA API returned error".into());
        }

        let odata: EsaODataResponse = resp.json().await.map_err(|e| e.to_string())?;

        let scenes = odata.value.into_iter().map(|p| SarScene {
            id: p.id,
            platform: "Sentinel-1".to_string(),
            date: p.content_date.start,
            footprint: p.footprint.unwrap_or_default(), // now populated
            quicklook_url: Some(format!(
                "https://catalogue.dataspace.copernicus.eu/odata/v1/Products('{}')/Products('Quicklook')/$value",
                p.id
            )),
        }).collect();

        Ok(scenes)
    }
}
